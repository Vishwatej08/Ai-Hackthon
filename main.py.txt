import threading
import time
import cv2
import mediapipe as mp
import pyautogui
import speech_recognition as sr
import numpy as np

# Disable PyAutoGUI fail-safe for demo purposes
pyautogui.FAIL_SAFE = False
screen_w, screen_h = pyautogui.size()

# Initialize MediaPipe Hands
mp_hands = mp.solutions.hands
hands = mp_hands.Hands(
    static_image_mode=False,
    max_num_hands=1,
    min_detection_confidence=0.7,
    min_tracking_confidence=0.7,
)
mp_draw = mp.solutions.drawing_utils

# Control State Variables
control_active = True
last_click_time = 0
click_cooldown = 0.5


def listen_commands():
  """Background thread to listen for voice commands."""
  global control_active
  recognizer = sr.Recognizer()
  mic = sr.Microphone()

  with mic as source:
    recognizer.adjust_for_ambient_noise(source, duration=1)

  print('[INFO] Voice command listener started...')

  while True:
    try:
      with mic as source:
        audio = recognizer.listen(source, timeout=5, phrase_time_limit=3)
      command = recognizer.recognize_google(audio).lower()
      print(f'[VOICE] Heard: {command}')

      if 'stop control' in command or 'stop' in command:
        control_active = False
        print('[STATUS] Gesture & Voice Control PAUSED.')
      elif 'start control' in command or 'resume' in command:
        control_active = True
        print('[STATUS] Gesture & Voice Control ACTIVE.')
      elif 'open browser' in command:
        pyautogui.hotkey('ctrl', 'shift', 'n')
      elif 'scroll down' in command:
        pyautogui.scroll(-300)
      elif 'scroll up' in command:
        pyautogui.scroll(300)

    except sr.WaitTimeoutError:
      continue
    except Exception:
      continue


# Start the voice command listener in a separate thread
voice_thread = threading.Thread(target=listen_commands, daemon=True)
voice_thread.start()

# Initialize Webcam
cap = cv2.VideoCapture(0)
cap.set(3, 640)
cap.set(4, 480)

print('[INFO] Starting webcam feed. Press \'q\' to exit.')

while cap.isOpened():
  success, frame = cap.read()
  if not success:
    print('[WARNING] Ignoring empty camera frame.')
    continue

  frame = cv2.flip(frame, 1)
  h, w, c = frame.shape

  rgb_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
  result = hands.process(rgb_frame)

  if result.multi_hand_landmarks and control_active:
    for hand_landmarks in result.multi_hand_landmarks:
      mp_draw.draw_landmarks(frame, hand_landmarks, mp_hands.HAND_CONNECTIONS)

      landmarks = hand_landmarks.landmark
      index_tip = np.array([landmarks[8].x * w, landmarks[8].y * h])
      thumb_tip = np.array([landmarks[4].x * w, landmarks[4].y * h])

      # Cursor Movement
      screen_x = np.interp(index_tip[0], (50, w - 50), (0, screen_w))
      screen_y = np.interp(index_tip[1], (50, h - 50), (0, screen_h))
      pyautogui.moveTo(screen_x, screen_y)

      # Pinch Detection (Click)
      distance = np.linalg.norm(index_tip - thumb_tip)
      cv2.putText(
          frame,
          f'Dist: {int(distance)}',
          (10, 50),
          cv2.FONT_HERSHEY_SIMPLEX,
          1,
          (0, 255, 0),
          2,
      )

      if distance < 35:
        current_time = time.time()
        if current_time - last_click_time > click_cooldown:
          pyautogui.click()
          last_click_time = current_time
          cv2.putText(
              frame,
              'PINCH CLICK!',
              (50, 100),
              cv2.FONT_HERSHEY_SIMPLEX,
              1,
              (0, 0, 255),
              2,
          )

  status_text = 'Status: ACTIVE' if control_active else 'Status: PAUSED'
  color = (0, 255, 0) if control_active else (0, 0, 255)
  cv2.putText(
      frame,
      status_text,
      (10, h - 20),
      cv2.FONT_HERSHEY_SIMPLEX,
      0.8,
      color,
      2,
  )

  cv2.imshow('AI Gesture & Voice Control', frame)

  if cv2.waitKey(1) & 0xFF == ord('q'):
    break

cap.release()
cv2.destroyAllWindows()
